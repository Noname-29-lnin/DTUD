\answer{a}{Giả sử OPAMP lí tưởng. Tìm hệ số khuếch đại $\dfrac{V_{o1}}{V_{in}}$, $\dfrac{V_{o2}}{V_{o1}}$ và $\dfrac{V_{o2}}{V_{in}}$.}

\begin{itemize}[label=-]
	\item Xét Op-Amp 1:
	\[ \frac{v_{o1}}{10R} + \frac{v_{o2}}{10R} = \frac{-V_{in}}{R} \quad (1) \]
	
	\item Xét Op-Amp 2:
	\[ \frac{v_{o2}}{2R} = \frac{v_{o1}}{R} \Rightarrow \frac{v_{o2}}{v_{o1}} = 2 \quad (2) \]
	
	\item Từ (1) và (2), ta có:
	\[ \frac{v_{o1}}{10R} + \frac{v_{o2}}{10R} = \frac{-V_{in}}{R} \Rightarrow 3v_{o1} = -10V_{in} \quad (3)\]
	\[ \Rightarrow \left\{ \frac{v_{o1}}{V_{in}} = -\frac{10}{3},\ \frac{v_{o2}}{V_{in}} = -\frac{20}{3} \right. \]
\end{itemize}

\answer{b}{Giả sử cả 2 OPAMP có $V_{io} = 12\,\mu\textsf{V}$, $I_{io} = 12\,\textsf{nA}$, $I_{ib} = 20\,\textsf{nA}$. Tìm ảnh hưởng của $V_{io}$ lên ngõ ra $V_{o2}$. Tìm giá trị của $R$ để có thể giảm thiếu tối đa ảnh hưởng của dòng $I_{ib}$ và $I_{io}$.}

\begin{itemize}[label=-]
	\item Xét ảnh hưởng của $V_{io}$:  
	Khi $V_{IN}=0,\ V_{io}\neq0$ và $I_{+}=I_{-}=0$  
	
	\begin{itemize}[label=+]
		\item Phương trình KCL tại nút (1):
		\[
		\frac{V_{io}}{R}+\frac{V_{io}-V_{o1}}{10R}+\frac{V_{io}-V_{o2}}{10R}=0
		\]
		\[
		\Rightarrow 12V_{io}=V_{o1}+V_{o2}
		\Rightarrow V_{o1}=12V_{io}-V_{o2}
		\]
		
		\item Phương trình KCL tại nút (2):
		\[
		\frac{(V_{o1}-V_{io})-V_{o2}}{R}+\frac{V_{o1}-V_{io}}{R}=0
		\]
		\[
		\Rightarrow 2V_{o1}-2V_{io}-V_{o2}=0
		\Rightarrow V_{o2}-2(12V_{io}-V_{o2})=2V_{io}
		\]
		\[
		\Rightarrow V_{o2}=\frac{22}{3}V_{io}
		\Rightarrow \Delta V_{o2}=\frac{22}{3}V_{io}
		\]
	\end{itemize}
	
	\item Xét ảnh hưởng của $I_{io}$ và $I_{ib}$: Khi $V_{IN}=0,\ V_{io}=0$ và $I_{+}\neq I_{-}\neq0$ 
	
	\[
	\left\{
	\begin{matrix}
		| I_{+} - I_{-} | = I_{io} 	\\[4pt]
		\dfrac{I_{+} + I_{-}}{2} = I_{ib}
	\end{matrix}
	\right.
	\Rightarrow
	\left\{
	\begin{matrix}
		I_{+} = 14\,\textsf{nA}	\\[4pt]
		I_{-} = 26\,\textsf{nA}
	\end{matrix}
	\right.
	\]
%	$I_{+} - I_{-} = I_{io}$, $I_{+} + I_{-} = 2I_{ib} 
%	\Rightarrow I_{+} = 14\,\text{nA},\ I_{-} = 26\,\text{nA}$ 
	
	\begin{itemize}[label=+]
		\item Phương trình KCL tại nút (1):
		\[
		I_{-}+\frac{0-V_{o1}}{10R}+\frac{0-V_{o2}}{10R}=0
		\Rightarrow I_{-}\cdot10R=V_{o1}+V_{o2}
		\]
		
		\item Phương trình KCL tại nút (2):
		\[
		I_{-}+\frac{V_{o1}}{R}+\frac{V_{o1}-V_{o2}}{R}=0
		\Rightarrow V_{o1}=\frac{V_{o2}-I_{-}R}{2}
		\]
		\[
		\Rightarrow I_{-} 10R=\frac{V_{o2}-I_{-}R}{2}+V_{o2}
		\Rightarrow V_{o2}=7I_{-}R
		\Rightarrow \Delta V_{o2}=7I_{-}R
		\]
	\end{itemize}
